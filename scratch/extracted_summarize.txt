654: def line_at_or_after(handle, offset: int, data_start: int) -> Tuple[int, int, str]:
655:     if offset <= data_start:
656:         handle.seek(data_start)
657:     else:
658:         handle.seek(offset)
659:         handle.readline()
660: 
661:     line_start = handle.tell()
662:     raw = handle.readline()
663:     line_end = handle.tell()
664:     if not raw:
665:         return line_start, line_end, ""
666:     return line_start, line_end, raw.decode("utf-8", errors="replace").strip()

668: def seek_first_timestamp_offset(path: Path, target: pd.Timestamp, data_start: int, time_index: int, delimiter: str) -> int:
669:     file_size = get_file_size(path)
670:     low = data_start
671:     high = file_size
672: 
673:     # Ensure target is timezone-naive UTC for safe comparison
674:     target_naive = target.tz_convert("UTC").tz_localize(None) if target.tzinfo is not None else target
675: 
676:     with open_compressed_file(path, "rb") as handle:
677:         for _ in range(80):
678:             if high - low <= 256:
679:                 break
680:             midpoint = (low + high) // 2
681:             line_start, line_end, line = line_at_or_after(handle, midpoint, data_start)
682:             if not line:
683:                 high = midpoint
684:                 continue
685: 
686:             values = csv_row_values(line, delimiter)
687:             row_time = parse_csv_time_value(values[time_index]) if len(values) > time_index else None
688:             # Both row_time and target_naive are now timezone-naive UTC
689:             if row_time is None or row_time < target_naive:
690:                 low = max(line_end, midpoint + 1)
691:             else:
692:                 high = line_start if line_start < high else midpoint
693: 
694:         aligned_start, _, _ = line_at_or_after(handle, low, data_start)
695:         return max(data_start, aligned_start)
