697: async function checkCacheLookup() {
698:     const panel = document.getElementById("cacheReuseCard");
699:     if (!panel) return;

701:         panel.classList.add("hidden");

706: 
707:     try {
708:         const resp = await fetch("/api/cache/lookup", {
709:             method: "POST",
710:             headers: { "Content-Type": "application/json" },
711:             body: JSON.stringify({
712:                 csv_path:       csvPathInput.value.trim(),
713:                 start_utc:      selectedStartUtc,

719:                 chart_pips:     chartPips,
720:             }),
721:         });
722:         if (!resp.ok) return;
723:         const data = await resp.json();
724:         _lastCacheLookup = data;
725:         renderCacheLookupPanel(data);
